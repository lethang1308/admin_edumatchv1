Table users {
  Note: 'Bảng xác thực chung cho bốn role; không trộn trường pháp nhân vào tài khoản.'
  id bigint [pk, increment, note: "Khóa chính; PK, AUTO_INCREMENT"]
  role varchar(20) [not null, default: 'student', note: "Vai trò: student | teacher | center | super_admin; NOT NULL; DEFAULT student"]
  name varchar(150) [not null, note: "Tên hiển thị hoặc tên người đại diện; NOT NULL"]
  email varchar(191) [unique, note: "Email đăng nhập; NULL; UNIQUE"]
  phone varchar(20) [unique, note: "SĐT đăng nhập, chuẩn hóa E.164; NULL; UNIQUE"]
  password varchar(255) [not null, note: "Mật khẩu đã băm; NOT NULL"]
  avatar_path varchar(255) [note: "Ảnh đại diện tài khoản; NULL"]
  status varchar(20) [not null, default: 'active', note: "active | blocked; NOT NULL; DEFAULT active"]
  email_verified_at datetime [note: "Thời điểm xác thực email; NULL"]
  phone_verified_at datetime [note: "Thời điểm xác thực SĐT; NULL"]
  last_login_at datetime [note: "Đăng nhập lần cuối; NULL"]
  blocked_reason text [note: "Lý do khóa tài khoản; NULL"]
  blocked_at datetime [note: "Thời điểm khóa tài khoản; NULL"]
  created_at datetime [not null, note: "Thời điểm tạo; NOT NULL"]
  updated_at datetime [not null, note: "Thời điểm cập nhật; NOT NULL"]
  deleted_at datetime [note: "Thời điểm xóa mềm; NULL"]
  indexes {
    (role, status, created_at)
  }
}
